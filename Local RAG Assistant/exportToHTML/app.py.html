37   def extract_documents(uploaded_files):
38       pages = []
39       for uploaded_file in uploaded_files:
40           uploaded_file.seek(0)
41           source = uploaded_file.name
42           if source.lower().endswith(".pdf"):
43               reader = PdfReader(uploaded_file)
44               for page_number, page in enumerate(reader.pages, start=1):
45                   text = (page.extract_text() or "").strip()
46                   if text:
47                       pages.append({"source": source, "page": page_number, "text": text})
48           else:
49               text = uploaded_file.getvalue().decode("utf-8", errors="replace").strip()
50               if text:
51                   pages.append({"source": source, "page": 1, "text": text})
52       return pages

55   def chunk_pages(pages):
56       chunks = []
57       for page in pages:
58           text = page["text"]
59           start = 0
60           chunk_index = 0
61           while start < len(text):
62               end = min(start + CHUNK_SIZE, len(text))
63               content = text[start:end].strip()
64               if content:
65                   chunks.append({
66                       "id": f"{page['source']}|{page['page']}|{chunk_index}",
67                       "source": page["source"],
68                       "page": page["page"],
69                       "chunk_index": chunk_index,
70                       "content": content,
71                   })
72               if end == len(text):
73                   break
74               start = end - CHUNK_OVERLAP
75               chunk_index += 1
76       return chunks

79   def literal_phrase_search(chunks, question):
80       normalized_question = question.strip().lower()
81       if not normalized_question:
82           return []
83       return [chunk for chunk in chunks if normalized_question in chunk["content"].lower()]

86   def store_chunks(chunks):
87       texts = [chunk["content"] for chunk in chunks]
88       embeddings = ollama.embed(model=EMBED_MODEL, input=texts).embeddings
89       rows = [
90           (
91               chunk["id"],
92               chunk["source"],
93               chunk["page"],
94               chunk["chunk_index"],
95               chunk["content"],
96               json.dumps(embedding),
97           )
98           for chunk, embedding in zip(chunks, embeddings)
99       ]
100      sources = sorted({chunk["source"] for chunk in chunks})
101      connection = sqlite3.connect(DB_PATH)
102      try:
103          connection.executemany(
104              "DELETE FROM chunks WHERE source = ?",
105              [(source,) for source in sources],
106          )
107          connection.executemany(
108              "INSERT INTO chunks (id, source, page, chunk_index, content, embedding) VALUES (?, ?, ?, ?, ?, ?)",
109              rows,
110          )
111          connection.commit()
112      finally:
113          connection.close()

134  def cosine_similarity(left, right):
135      dot_product = sum(a * b for a, b in zip(left, right))
136      left_magnitude = sum(value * value for value in left) ** 0.5
137      right_magnitude = sum(value * value for value in right) ** 0.5
138      if left_magnitude == 0 or right_magnitude == 0:
139          return 0.0
140      return dot_product / (left_magnitude * right_magnitude)

143  def retrieve(question):
144      rows = load_rows()
145      if not rows:
146          return []
147      query_vector = ollama.embed(model=EMBED_MODEL, input=question).embeddings[0]
148      scored = []
149      for source, page, chunk_index, content, embedding_json in rows:
150          scored.append({
151              "source": source,
152              "page": page,
153              "chunk_index": chunk_index,
154              "content": content,
155              "score": cosine_similarity(query_vector, json.loads(embedding_json)),
156          })
157      scored.sort(key=lambda item: item["score"], reverse=True)
158      return scored[:TOP_K]

161  def answer_question(question, evidence):
162      evidence_text = "\n\n".join(
163          f"[Source {index}: {item['source']}, page {item['page']}]\n{item['content']}"
164          for index, item in enumerate(evidence, start=1)
165      )
166      response = ollama.chat(
167          model=CHAT_MODEL,
168          messages=[
169              {
170                  "role": "system",
171                  "content": "You are a careful research assistant. Answer only from the provided evidence. Cite claims with the exact bracketed source labels. If the evidence is insufficient, say so clearly.",
172              },
173              {
174                  "role": "user",
175                  "content": f"Question:\n{question}\n\nEvidence:\n{evidence_text}",
176              },
177          ],
178      )
179      return response["message"]["content"]
